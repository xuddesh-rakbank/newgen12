				UpdateCustomFlags.updateAccountSummaryFlag(iform, wi_name, rowIndex, customer_type, prodFetchType);
				String result = "Success" + "_" + rowIndex + "_" + customer_type + "_" + prodFetchType + "_" + cif;
				CourtOrder.mLogger.debug("result in account summary is :" + result);
				return result;

			}

			else {
				CourtOrder.mLogger.debug("Error in account summary: ");
				// String result =
				// "Success"+"_"+rowIndex+"_"+customer_type+"_"+prodFetchType;
				// //hardcoded
				return "error";
			}

		} catch (Exception e) {
			String exception = customException(e);
			CourtOrder.mLogger.debug("Exception in execute_AccountSummary :" + exception);
		}
		return "";
	}
